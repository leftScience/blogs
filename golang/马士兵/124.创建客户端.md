【1】调用Dial函数：（net包下）

![](/media/knowledge/articles/local/go-notes-126/56363d00c7ba40dc-image.png)

【2】代码：

```go
package main

import(
        "fmt"
        "net" //所需的网络编程全部都在net包下
)

func main(){
        //打印：
        fmt.Println("客服端启动。。")
        //调用Dial函数：参数需要指定tcp协议，需要指定服务器端的IP+PORT
        conn,err := net.Dial("tcp","127.0.0.1:8888")
        if err != nil {//连接失败
                fmt.Println("客户端连接失败：err:",err)
                return
        }
        fmt.Println("连接成功，conn:",conn)
}
```
