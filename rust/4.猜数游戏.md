```
use rand::Rng;
use std::{cmp::Ordering, io};

fn main() {
    println!("猜数游戏");

    // 创建一个线程本地的随机数生成器
    let mut rng = rand::rng();

    // 1. 生成一个 1 到 100 之间的随机整数（包含 1 和 100）
    let secret_num: u32 = rng.random_range(1..=100);

    println!("神秘数字是{} \n", secret_num);

    loop {
        println!("请输入一个数字");

        let mut guess = String::new();

        io::stdin().read_line(&mut guess).expect("无法读取行");

        // 把字符串转化为整数
        // let guess: u32 = guess.trim().parse().expect("转换失败");
        let guess: u32 = match guess.trim().parse() {
            Ok(num) => num,
            Err(_) =>{
                println!("输入的内容不合法");
                continue
            },
        };

        match guess.cmp(&secret_num) {
            Ordering::Less => println!("少了"),
            Ordering::Greater => println!("多了"),
            Ordering::Equal => {
                println!("猜对了");
                break;
            }
        }
    }
}
```
